cmake_minimum_required(VERSION 3.24)
project(LinkBridgeWindowCapture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(LinkBridgeWindowCapture SHARED WindowCapture.cpp)
add_executable(LinkBridgeWindowCaptureProbe CaptureProbe.cpp)
add_executable(LinkBridgeWindowRoamingFixture WIN32 RoamingFixture.cpp)
foreach(target LinkBridgeWindowCapture LinkBridgeWindowCaptureProbe LinkBridgeWindowRoamingFixture)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  target_include_directories(${target} PRIVATE "${CMAKE_WINDOWS_KITS_10_DIR}/Include/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/cppwinrt")
  target_include_directories(${target} PRIVATE "C:/Program Files (x86)/Windows Kits/10/Include/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/cppwinrt")
endforeach()
target_link_libraries(LinkBridgeWindowCapture PRIVATE d3d11 dxgi runtimeobject windowsapp user32)
target_link_libraries(LinkBridgeWindowCaptureProbe PRIVATE LinkBridgeWindowCapture user32 gdi32 dwmapi)
target_link_libraries(LinkBridgeWindowRoamingFixture PRIVATE user32 gdi32)
set_target_properties(LinkBridgeWindowCapture PROPERTIES OUTPUT_NAME "LinkBridge.WindowCapture")
